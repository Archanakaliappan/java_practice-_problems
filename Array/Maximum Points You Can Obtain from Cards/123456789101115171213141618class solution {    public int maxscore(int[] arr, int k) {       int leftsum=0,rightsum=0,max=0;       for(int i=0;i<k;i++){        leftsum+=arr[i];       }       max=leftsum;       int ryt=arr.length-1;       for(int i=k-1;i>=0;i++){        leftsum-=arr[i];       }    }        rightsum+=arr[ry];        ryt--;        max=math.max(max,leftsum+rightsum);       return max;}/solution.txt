class Solution {
    public int maxScore(int[] arr, int k) {
       int leftsum=0,rightsum=0,max=0;
       for(int i=0;i<k;i++){
        leftsum+=arr[i];

       }
       max=leftsum;
       int ryt=arr.length-1;
       for(int i=k-1;i>=0;i++){
        leftsum-=arr[i];
       }
    }
        rightsum+=arr[ry];
        ryt--;
        max=Math.max(max,leftsum+rightsum);
       return max;
}
